// src/store/useAnalyticsStore.ts
import type { IPlan } from "@/types"
import { create } from "zustand"



interface AnalyticsState {
  subjectAssessmentPlan: IPlan[]
  subjectLessonPlan: IPlan[]
  studentConductCharacterPlan: IPlan[]
  studentConductCharacterAssessmentPlan: IPlan[]
  projectTaskPlan: IPlan[]
  projectTaskFacilitationPlan: IPlan[]

  filterSubjectAssessmentPlan: IPlan[],
  filterSubjectLessonPlan: IPlan[],
  filterStudentConductCharacterPlan: IPlan[],
  filterStudentConductCharacterAssessmentPlan: IPlan[],
  filterProjectTaskPlan: IPlan[],
  filterProjectTaskFacilitationPlan: IPlan[],

  setSubjectAssessmentPlan: (plans: IPlan[]) => void
  setSubjectLessonPlan: (plans: IPlan[]) => void
  setStudentConductCharacterPlan: (plans: IPlan[]) => void
  setStudentConductCharacterAssessmentPlan: (plans: IPlan[]) => void
  setProjectTaskPlan: (plans: IPlan[]) => void
  setProjectTaskFacilitationPlan: (plans: IPlan[]) => void
  getTotalTeachers: () => number

  // subject assessment view details variables
  filterSubjectAssessmentBySubject: string
  filterSubjectAssessmentByClass: string
  filterSubjectAssessmentByTerm: string
  // subject assessment view details filters
  setFilterSubjectAssessmentBySubject: (subject: string) => IPlan[]
  setFilterSubjectAssessmentByClass: (classLevel: string) => void
  setFilterSubjectAssessmentByTerm: (term: string) => void

  resetAllPlans: () => void

  subjectAssessments: string
  subjectLessonPlans: string
  conductCharacterLessonPlans: string
  conductCharacterAssessments: string
  projectTasks: string
  projectFacilitationPlans: string

  setSubjectAssessments: (value: string) => void
  setSubjectLessonPlans: (value: string) => void
  setConductCharacterLessonPlans: (value: string) => void
  setConductCharacterAssessments: (value: string) => void
  setProjectTasks: (value: string) => void
  setProjectFacilitationPlans: (value: string) => void


  // NEW: Add these for select filters
  subject: string
  classLevel: string
  term: string
  date: string

  setSubject: (value: string) => void
  setClassLevel: (value: string) => void
  setTerm: (value: string) => void
  setDate: (value: string) => void
}

export const useAnalyticsStore = create<AnalyticsState>((set, get) => ({
  // original values
  subjectAssessmentPlan: [],
  subjectLessonPlan: [],
  studentConductCharacterPlan: [],
  studentConductCharacterAssessmentPlan: [],
  projectTaskPlan: [],
  projectTaskFacilitationPlan: [],
  // filters values
  filterSubjectAssessmentPlan: [],
  filterSubjectLessonPlan: [],
  filterStudentConductCharacterPlan: [],
  filterStudentConductCharacterAssessmentPlan: [],
  filterProjectTaskPlan: [],
  filterProjectTaskFacilitationPlan: [],
  // filter states
  subjectAssessments: "",
  subjectLessonPlans: "",
  conductCharacterLessonPlans: "",
  conductCharacterAssessments: "",
  projectTasks: "",
  projectFacilitationPlans: "",

  // subject assessment view details variables
  filterSubjectAssessmentBySubject: "",
  filterSubjectAssessmentByClass: "",
  filterSubjectAssessmentByTerm: "",

  subject: "",
  classLevel: "",
  term: "",
  date: "",

  setSubject: (value) => set({ subject: value }),
  setClassLevel: (value) => set({ classLevel: value }),
  setTerm: (value) => set({ term: value }),
  setDate: (value) => set({ date: value }),

  // Setters
  setSubjectAssessments: (value) => set({ subjectAssessments: value }),
  setSubjectLessonPlans: (value) => set({ subjectLessonPlans: value }),
  setConductCharacterLessonPlans: (value) => set({ conductCharacterLessonPlans: value }),
  setConductCharacterAssessments: (value) => set({ conductCharacterAssessments: value }),
  setProjectTasks: (value) => set({ projectTasks: value }),
  setProjectFacilitationPlans: (value) => set({ projectFacilitationPlans: value }),

  setSubjectAssessmentPlan: (plans) => set(() => ({
    subjectAssessmentPlan: plans,
    filterSubjectAssessmentPlan: plans
  })),
  setSubjectLessonPlan: (plans) => set(() =>
  ({
    subjectLessonPlan: plans,
    filterSubjectLessonPlan: plans
  })),
  setStudentConductCharacterPlan: (plans) => set(() =>
  ({
    studentConductCharacterPlan: plans,
    filterStudentConductCharacterPlan: plans
  })),
  setStudentConductCharacterAssessmentPlan: (plans) =>
    set(() => ({
      studentConductCharacterAssessmentPlan: plans
      , filterStudentConductCharacterAssessmentPlan: plans
    })),
  setProjectTaskPlan: (plans) => set(() =>
    ({ projectTaskPlan: plans, filterProjectTaskPlan: plans })),
  setProjectTaskFacilitationPlan: (plans) => set(() =>
  ({
    projectTaskFacilitationPlan: plans,
    filterProjectTaskFacilitationPlan: plans
  })),

  getTotalTeachers: () => {
    const source = [
      ...get().subjectAssessmentPlan,
      ...get().subjectLessonPlan,
      ...get().studentConductCharacterPlan,
      ...get().studentConductCharacterAssessmentPlan,
      ...get().projectTaskPlan,
      ...get().projectTaskFacilitationPlan,
    ];
    return new Set(source.map((item) => item.userId.username)).size;
  },

  setFilterSubjectAssessmentBySubject: (subject) => {
    const filtered = get().subjectAssessmentPlan.filter(
      (plan) => plan.fields?.[12] === subject
    )

    set(() => ({
      filterSubjectAssessmentPlan: filtered,
      filterSubjectAssessmentBySubject: subject
    })) // update filtered version
    return filtered
  },

  setFilterSubjectAssessmentByClass: (classLevel) => {
    const filtered = get().subjectAssessmentPlan.filter(
      (plan) => plan.fields?.[4] === classLevel
    )
    set(() => ({
      filterSubjectAssessmentPlan: filtered,
      filterSubjectAssessmentByClass: classLevel
    }))
  },
  setFilterSubjectAssessmentByTerm: (term) => {
    const filtered = get().subjectAssessmentPlan.filter(
      (plan) => plan.fields?.[10] === term
    )
    set(() => ({ filterSubjectAssessmentPlan: filtered }))
  },

  resetAllPlans: () => {
    set((state) => ({
      filterSubjectAssessmentPlan: state.subjectAssessmentPlan,
      filterSubjectLessonPlan: state.subjectLessonPlan,
      filterStudentConductCharacterPlan: state.studentConductCharacterPlan,
      filterStudentConductCharacterAssessmentPlan: state.studentConductCharacterAssessmentPlan,
      filterProjectTaskPlan: state.projectTaskPlan,
      filterProjectTaskFacilitationPlan: state.projectTaskFacilitationPlan,
      subject: "",
      classLevel: "",
      term: "",
      date: "",
    }))
  }
}))
